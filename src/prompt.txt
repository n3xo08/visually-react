/* 1. Define the base stacking container */
.uslugi-stacking-container {
  display: grid;
  place-items: center;
  gap: 0; /* Cards must touch to stack */
  grid-template-columns: 1fr;
}

/* 2. Configure the View Timeline on the entire section */
.uslugi-section {
  view-timeline-name: --stacking-cards;
  view-timeline-axis: block;
}

/* 3. Base style for the Cards (Already look good, maintain glassmorphism) */
.uslugi-glass-card {
  --start-scroll: 20%;
  --end-scroll: 80%;
  
  /* Apply animation linked to the timeline */
  animation: stack-and-fade linear both;
  animation-timeline: --stacking-cards;
  
  /* Use Tailwind for dynamic range (ensure these classes exist) */
  animation-range: entry calc(var(--start-scroll)) exit calc(var(--end-scroll));
  
  /* Ensure 3D context for stacking */
  transform-style: preserve-3d;
  backface-visibility: hidden;
}

/* 4. The Animation Logic (3D Stack + Smooth Fade) */
@keyframes stack-and-fade {
  from {
    /* Entered View: Opaque, flat, and slides up */
    transform: translateY(100px) scale(0.9) translateZ(0);
    opacity: 1;
  }
  to {
    /* Exiting View: Fully stacked, slightly transparent, scaled down, and translated deeper in 3D */
    transform: translateY(0px) scale(0.95) translateZ(-50px);
    opacity: 0.1; /* Keep slightly visible for the effect */
  }
}

/* 5. Handle Stacking Index (Required for 'on top of' effect) */
.uslugi-glass-card:nth-child(1) { z-index: 10; }
